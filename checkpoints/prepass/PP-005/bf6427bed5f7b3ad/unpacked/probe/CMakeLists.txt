cmake_minimum_required(VERSION 3.15)
project(PP005Probe LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(PlanePathLua 5.5.1 EXACT CONFIG REQUIRED PATHS "${PLANEPATH_LUA_PREFIX}" NO_DEFAULT_PATH)
if(NOT PlanePathLua_REVISION STREQUAL "5.5.1-pp1")
    message(FATAL_ERROR "Unexpected PlanePathLua revision")
endif()
add_executable(pp005_probe probe.cpp)
target_compile_features(pp005_probe PRIVATE cxx_std_17)
target_compile_definitions(pp005_probe PRIVATE PP_USE_UPSTREAM_HEADERS)
target_link_libraries(pp005_probe PRIVATE PlanePathLua::Lua Threads::Threads)
enable_testing()
add_test(NAME pp005_mechanisms COMMAND pp005_probe "${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
set_tests_properties(pp005_mechanisms PROPERTIES TIMEOUT 15)
