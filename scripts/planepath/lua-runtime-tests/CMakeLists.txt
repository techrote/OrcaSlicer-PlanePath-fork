cmake_minimum_required(VERSION 3.15)
project(PlanePathLuaRuntimeTests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT DEFINED PLANEPATH_LUA_PREFIX)
    message(FATAL_ERROR "PLANEPATH_LUA_PREFIX must identify the exact installed Orca Lua dependency")
endif()
find_package(PlanePathLua 5.5.1 EXACT CONFIG REQUIRED
    PATHS "${PLANEPATH_LUA_PREFIX}/lib/cmake/PlanePathLua" NO_DEFAULT_PATH)
if(NOT PlanePathLua_REVISION STREQUAL "5.5.1-pp1")
    message(FATAL_ERROR "Unexpected Lua dependency profile")
endif()
get_target_property(_kind PlanePathLua::Lua TYPE)
if(NOT _kind STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "PP-005 requires static Lua")
endif()
set(_src "${CMAKE_CURRENT_LIST_DIR}/../../../src/libslic3r/Script")
add_library(planepath_script_runtime STATIC "${_src}/ScriptRuntime.cpp" "${_src}/ScriptRuntime.hpp")
target_include_directories(planepath_script_runtime PUBLIC "${_src}")
target_link_libraries(planepath_script_runtime PRIVATE PlanePathLua::Lua)
find_package(Threads REQUIRED)
add_executable(planepath_lua_runtime_tests main.cpp)
target_link_libraries(planepath_lua_runtime_tests PRIVATE planepath_script_runtime Threads::Threads)
add_test(NAME planepath_lua_sandbox COMMAND planepath_lua_runtime_tests)
set_tests_properties(planepath_lua_sandbox PROPERTIES TIMEOUT 90 LABELS "ScriptRuntime;Security")
